cmake_minimum_required(VERSION 3.16)
project(minigit VERSION 1.0.0 LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MINIGIT_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(MINIGIT_FETCH_GTEST "Download GoogleTest if not installed" ON)
add_library(minigit_core
    src/sha1.cpp src/io.cpp src/object.cpp src/store.cpp
    src/diff.cpp src/repository.cpp src/command.cpp)
target_include_directories(minigit_core PUBLIC include)
if(MSVC)
  target_compile_options(minigit_core PRIVATE /W4)
else()
  target_compile_options(minigit_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINIGIT_SANITIZERS AND NOT MSVC)
  target_compile_options(minigit_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(minigit_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(minigit src/main.cpp)
target_link_libraries(minigit PRIVATE minigit_core)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest CONFIG QUIET)
  if(NOT GTest_FOUND)
    if(NOT MINIGIT_FETCH_GTEST)
      message(FATAL_ERROR "Install GoogleTest or enable MINIGIT_FETCH_GTEST")
    endif()
    include(FetchContent)
    FetchContent_Declare(googletest
      URL https://github.com/google/googletest/archive/03597a01ee50ed33e9dfd640b249b4be3799d395.zip
      URL_HASH SHA256=edd885a1ab32b6999515a880f669efadb80b3f880215f315985fa3f6eca7c4d3)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
  endif()
  add_executable(minigit_tests tests/core_test.cpp tests/repository_test.cpp tests/command_test.cpp)
  target_link_libraries(minigit_tests PRIVATE minigit_core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(minigit_tests)
endif()
install(TARGETS minigit RUNTIME DESTINATION bin)
